def words_per_line(name):
    counts = []
    with open(name) as f:
        for line in f:
            if line.strip():
                counts.append(len(line.split()))
    return counts

@@solution
import csv

def methods(name):
    counts = {}
    with open(name, newline="") as f:
        for row in csv.DictReader(f):
            m = row["method"].strip().lower()
            counts[m] = counts.get(m, 0) + 1
    return counts

def parse_amount(text):
    cleaned = text.replace(",", "").strip()
    if not cleaned:
        raise ValueError("no amount")
    digits = cleaned.removeprefix("-").replace(".", "", 1)
    if not digits.isdigit():
        raise ValueError(f"not an amount: {text!r}")
    return float(cleaned)

## Forge: the data cleaner

Look at `payments.csv` in the box above. Some rows are fine. Others have problems: a date written the other way round, a missing amount, a negative payment, a duplicate, a date that doesn't exist, an amount far too large to be real, names and methods in odd cases with stray spaces. Your job is to build the program that a business would run every night on files like this one.

:::exercise id=forge points=50 title="Forge: the data cleaner"
Write a program, built from these functions, that cleans a payments file.

- `clean_row(row)` takes one row as a dictionary from `DictReader` and returns a new dictionary with:
  - `id`: an int;
  - `date`: the date as text in the form `YYYY-MM-DD`, which must be a real date in 2026;
  - `payer`: stripped, with each word capitalised (`"  bo musa "` gives `"Bo Musa"`);
  - `amount`: a float from 0.01 to 1,000,000, which may contain commas;
  - `method`: one of `card`, `cash`, `transfer`, `cheque`, in lower case.

  If any field is bad, it raises a `ValueError` whose message names the problem: `bad date '01/10/2026'`, `no amount`, `bad amount '-500'`, `bad method 'bitcoin'`, and so on (the exact wording of the messages is up to you, but the message must include the bad value in quotes, except for `no amount`).
- `clean_file(source)` reads the file and returns a tuple `(good, problems)`. `good` is the list of clean rows, in order, **leaving out duplicates**: a row with the same date, payer and amount as an earlier good row. `problems` is a list of `(id, message)` tuples, one per rejected row, with the message `duplicate of N` for a duplicate (where N is the earlier row's id). The `id` in a problem is the text from the file.
- `write_outputs(good, problems)` writes three files:
  - `clean.csv`: the good rows, with the header `id,date,payer,amount,method` and amounts written with 2 decimal places, like `12500.00`;
  - `problems.txt`: one line per problem, `row 3: bad date '01/10/2026'`;
  - `summary.txt`: exactly these lines:

```
Rows read: 12
Accepted: 6
Rejected: 6
Total accepted: 28,550.50
```

  (with the numbers for the file read, and the total of accepted amounts with comma separators and 2 places).
- The main program cleans `payments.csv`, writes the outputs, and prints the contents of `summary.txt`.
@@starter
import csv

METHODS = {"card", "cash", "transfer", "cheque"}

@@tests
import csv as _csv
def got():

@@hint

@@hint

@@solution
import csv

METHODS = {"card", "cash", "transfer", "cheque"}
DAYS = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]

def clean_date(text):
    parts = text.strip().split("-")
    if [len(p) for p in parts] != [4, 2, 2] or not all(p.isdigit() for p in parts):
        raise ValueError(f"bad date {text!r}")
    year, month, day = (int(p) for p in parts)
    if year != 2026 or not 1 <= month <= 12 or not 1 <= day <= DAYS[month - 1]:
        raise ValueError(f"bad date {text!r}")
    return text.strip()

def clean_amount(text):
    cleaned = text.replace(",", "").strip()
    if not cleaned:
        raise ValueError("no amount")
    digits = cleaned.removeprefix("-").replace(".", "", 1)
    if not digits.isdigit():
        raise ValueError(f"bad amount {text!r}")
    amount = float(cleaned)
    if not 0.01 <= amount <= 1_000_000:
        raise ValueError(f"bad amount {text!r}")
    return amount

def clean_method(text):
    method = text.strip().lower()
    if method not in METHODS:
        raise ValueError(f"bad method {text!r}")
    return method

def clean_row(row):
    return {
        "id": int(row["id"]),
        "date": clean_date(row["date"]),
        "payer": " ".join(w.capitalize() for w in row["payer"].split()),
        "amount": clean_amount(row["amount"]),
        "method": clean_method(row["method"]),
    }

def clean_file(source):
    good, problems, seen = [], [], {}
    with open(source, newline="") as f:
        for row in csv.DictReader(f):
            try:
                clean = clean_row(row)
            except ValueError as e:
                problems.append((row["id"], str(e)))
                continue
            key = (clean["date"], clean["payer"], clean["amount"])
            if key in seen:
                problems.append((row["id"], f"duplicate of {seen[key]}"))
                continue
            seen[key] = clean["id"]
            good.append(clean)
    return good, problems

def write_outputs(good, problems):
    with open("clean.csv", "w", newline="") as f:
        writer = csv.writer(f)
        writer.writerow(["id", "date", "payer", "amount", "method"])
        for r in good:
            writer.writerow([r["id"], r["date"], r["payer"], f"{r['amount']:.2f}", r["method"]])
    with open("problems.txt", "w") as f:
        for row_id, message in problems:
            f.write(f"row {row_id}: {message}\n")
    total = sum(r["amount"] for r in good)
    with open("summary.txt", "w") as f:
        f.write(f"Rows read: {len(good) + len(problems)}\n")
        f.write(f"Accepted: {len(good)}\n")
        f.write(f"Rejected: {len(problems)}\n")
        f.write(f"Total accepted: {total:,.2f}\n")

good, problems = clean_file("payments.csv")
write_outputs(good, problems)
with open("summary.txt") as f:
    print(f.read(), end="")
:::

:::think title="Never silently drop data"
A cleaner that quietly skipped bad rows would be dangerous: a payment would vanish and nobody would know. Yours rejects bad rows **loudly**, with a reason for each, in a file someone can check and fix. That's the difference between a script and a tool people can trust. Notice too that `clean_row` never prints or decides what to do with a bad row: it raises, and `clean_file` decides. Each function has one job.
:::

## Forge bonus: saving safely

If a program crashes, or the power goes off, halfway through writing a file, the file is left half-written and the old version is already gone. Careful programs avoid this. They write to a **temporary file** first, and only when that's complete do they swap it into place. Many also keep a few **backups**.

:::exercise id=forge2 points=30 title="Forge bonus: saving safely"
Write `safe_save(name, text, keep=3)` that:

1. writes `text` to a temporary file in the same folder, named like the target with `.tmp` added (`notes.txt.tmp`);
2. if the target already exists, shifts the backups along: `notes.txt.2` becomes `notes.txt.3`, `notes.txt.1` becomes `notes.txt.2`, and the current `notes.txt` becomes `notes.txt.1`, keeping at most `keep` backups (the oldest is deleted);
3. renames the temporary file to the target name.

Also write `restore(name, n=1)` that copies backup `n` back over the file, raising a `FileNotFoundError` with the message `no backup N of NAME` if it doesn't exist.

Use `pathlib`: `Path.replace(target)` renames a file, replacing any file already there; `Path.unlink()` deletes one; `p.with_name(...)` gives a path with a different name in the same folder.
@@starter
from pathlib import Path

@@tests
def test_saves():
    "saving and backups"
    reset_files()
    for i in range(1, 6):
        safe_save("notes.txt", f"version {i}\n")
    assert read_file("notes.txt") == "version 5\n"
    assert read_file("notes.txt.1") == "version 4\n"
    assert read_file("notes.txt.2") == "version 3\n"
    assert read_file("notes.txt.3") == "version 2\n"
    assert read_file("notes.txt.4") is None, "only 3 backups should be kept"
    assert read_file("notes.txt.tmp") is None, "the temporary file should be gone"
def test_new_file():
    "a new file"
    safe_save("fresh.txt", "hello\n", keep=2)
    assert read_file("fresh.txt") == "hello\n" and read_file("fresh.txt.1") is None
def test_keep():
    "keep=1"
    safe_save("k.txt", "a\n", keep=1)
    safe_save("k.txt", "b\n", keep=1)
    safe_save("k.txt", "c\n", keep=1)
    assert read_file("k.txt") == "c\n" and read_file("k.txt.1") == "b\n" and read_file("k.txt.2") is None
def test_restore():
    "restore"
    reset_files()
    safe_save("notes.txt", "new\n")
    restore("notes.txt")
    assert read_file("notes.txt") == "first line\nsecond line\n"
    try:
        restore("notes.txt", 2)
    except FileNotFoundError as e:
        assert str(e) == "no backup 2 of notes.txt"
    else:
        raise AssertionError("restoring a missing backup should raise FileNotFoundError")
@@hint
`backup = lambda i: path.with_name(f"{path.name}.{i}")`. Shift from the oldest down: first delete backup `keep` if it exists, then for `i` from `keep - 1` down to 1, move backup `i` to `i + 1`, then move the current file to backup 1.
@@hint
For `restore`, `path.write_text(backup.read_text())` copies it back.
@@solution
from pathlib import Path

def backup(path, i):
    return path.with_name(f"{path.name}.{i}")

def safe_save(name, text, keep=3):
    path = Path(name)
    tmp = path.with_name(path.name + ".tmp")
    tmp.write_text(text)
    if path.exists():
        oldest = backup(path, keep)
        if oldest.exists():
            oldest.unlink()
        for i in range(keep - 1, 0, -1):
            if backup(path, i).exists():
                backup(path, i).replace(backup(path, i + 1))
        path.replace(backup(path, 1))
    tmp.replace(path)

def restore(name, n=1):
    path = Path(name)
    b = backup(path, n)
    if not b.exists():
        raise FileNotFoundError(f"no backup {n} of {name}")
    path.write_text(b.read_text())
