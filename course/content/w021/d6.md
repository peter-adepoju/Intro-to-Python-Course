def largest_files(folder, k):
    files = [(name, len(v)) for name, v in folder.items() if isinstance(v, str)]
    files.sort(key=lambda p: (-p[1], p[0]))
    return [name for name, size in files[:k]]

Write `main(argv, text)` for a tool used as `find PATTERN [-i] [-n]`, which returns the lines of `text` that contain `PATTERN`: with `-i` ignoring case, and with `-n` putting the line number and a colon in front (`3:line`). @@starter

@@tests

    "several"
    t = "Buy eggs\ncall Ada\nEggs again\n"
    assert main(["eggs"], t) == (0, ["Buy eggs"])

@@solution
import argparse

def main(argv, text):
    p = argparse.ArgumentParser(prog="find")
    p.add_argument("pattern")
    p.add_argument("-i", action="store_true")
    p.add_argument("-n", action="store_true")
    args = p.parse_args(argv)
    found = []
    for number, line in enumerate(text.splitlines(), start=1):
        hay, needle = (line.lower(), args.pattern.lower()) if args.i else (line, args.pattern)
        if needle in hay:
            found.append(f"{number}:{line}" if args.n else line)
    return (0 if found else 1), found

def test_values():
    "several"
    p = {"c1": [], "c2": ["c1"], "x1": ["c2"], "m": ["c2", "x1"], "y1": ["m"], "z1": ["c2"]}

@@solution
def merged_into(parents, branches, target):
    seen, stack = set(), [branches[target]]
    while stack:
        c = stack.pop()
        if c not in seen:
            seen.add(c)
            stack.extend(parents[c])
    return sorted(name for name, c in branches.items() if name != target and c in seen)

:::exercise id=mix_ignore points=10 title=".gitignore rules"
Write `ignored(path, rules)` for a list of `.gitignore` rules: a rule ending in `/` ignores that folder anywhere in the path (`"__pycache__/"` ignores `"src/__pycache__/x.pyc"`); any other rule is a wildcard pattern matched with `fnmatch` against the file's **name** (the last part of the path). A rule starting with `!` un-ignores matching files, and later rules win over earlier ones.
@@starter

@@tests
def test_values():
    "several"
    rules = ["__pycache__/", "*.log", "!keep.log", ".env"]
    assert ignored("src/__pycache__/x.pyc", rules) and ignored("app.log", rules) and ignored("logs/old.log", rules)
    assert not ignored("keep.log", rules) and not ignored("main.py", rules) and ignored(".env", rules)

@@solution
import fnmatch

def ignored(path, rules):
    parts = path.split("/")
    result = False
    for rule in rules:
        negate = rule.startswith("!")
        pattern = rule[1:] if negate else rule
        if pattern.endswith("/"):
            hit = pattern[:-1] in parts[:-1]
        else:
            hit = fnmatch.fnmatch(parts[-1], pattern)
        if hit:
            result = not negate
    return result

## Forge: snap, a tiny version control tool

`snap` keeps snapshots of the text files in the current folder, in a hidden folder called `.snap`. It's used like this:

```
snap init
snap commit -m "First draft"
snap status
snap diff
snap log
snap restore notes.txt
```

It stores everything in `.snap/store.json`: a dictionary with `"commits"` (commit ID to `{"message", "parent", "files"}`) and `"head"` (the newest commit's ID, or `null`). This page has two files to start with.

:::exercise id=forge points=50 title="Forge: snap, a tiny version control tool"
Write `main(argv)` returning `(exit_code, output_lines)`, using `argparse` subcommands and the files in the **current folder** (only plain files directly in it, not in sub-folders, and never anything in `.snap`):

- `init`: create `.snap/store.json` with no commits; output `Initialised snap`. If it already exists: exit code 1, output `Already a snap folder`.
- `commit -m MESSAGE`: snapshot every file's text. If nothing has changed since the head commit: exit 1, output `Nothing to commit`. Otherwise store a commit with an ID from `commit_id` (in the starter), make it the head, and output `[ID] MESSAGE`.
- `log`: one line `ID MESSAGE` per commit from the head back; if there are none, `No commits yet`.
- `status`: for each file that differs from the head commit, sorted by name, a line `new: NAME`, `modified: NAME` or `deleted: NAME`; if there's nothing, `Nothing changed`.
- `diff`: for each modified file, sorted, a line `--- NAME` then the changed lines from `difflib.ndiff` (those starting `- ` or `+ `).
- `restore NAME`: write the head commit's version of the file back; output `Restored NAME`; exit 1 with `No such file in the last commit: NAME` if it isn't there.

Every command except `init` gives exit code 1 and `Not a snap folder (run snap init)` if there's no store.
@@files
=== notes.txt
Buy eggs
Call Ada
=== todo.txt
Finish the Forge
@@starter
import argparse
import difflib
import hashlib
import json
import os

STORE = os.path.join(".snap", "store.json")

def commit_id(files, message, parent):
    lines = [f"parent {parent or 'none'}", f"message {message}"]
    for name in sorted(files):
        lines.append(f"file {name} {files[name]}")
    return hashlib.sha1("\n".join(lines).encode()).hexdigest()[:7]

Helpers make it manageable: `load()` and `save(store)` for the JSON file; `working()` returning `{name: text}` for every file in `os.listdir(".")` that `os.path.isfile` says is a file; `head_files(store)` returning the head commit's files (or `{}`).
@@hint
For `status`, compare the two dictionaries: names in working but not head are new; in both with different text, modified; in head but not working, deleted. Collect `(name, line)` pairs and sort by name.
@@solution
import argparse
import difflib
import hashlib
import json
import os

STORE = os.path.join(".snap", "store.json")

def commit_id(files, message, parent):
    lines = [f"parent {parent or 'none'}", f"message {message}"]
    for name in sorted(files):
        lines.append(f"file {name} {files[name]}")
    return hashlib.sha1("\n".join(lines).encode()).hexdigest()[:7]

def load():
    with open(STORE) as f:
        return json.load(f)

def save(store):
    with open(STORE, "w") as f:
        json.dump(store, f, indent=1)

def working():
    files = {}
    for name in sorted(os.listdir(".")):
        if os.path.isfile(name):
            with open(name) as f:
                files[name] = f.read()
    return files

def head_files(store):
    return store["commits"][store["head"]]["files"] if store["head"] else {}

def changes(store):
    old, new = head_files(store), working()
    found = []
    for name in set(old) | set(new):
        if name not in old:
            found.append((name, "new"))
        elif name not in new:
            found.append((name, "deleted"))
        elif old[name] != new[name]:
            found.append((name, "modified"))
    return sorted(found)

def make_parser():
    p = argparse.ArgumentParser(prog="snap")
    sub = p.add_subparsers(dest="command", required=True)
    sub.add_parser("init")
    c = sub.add_parser("commit")
    c.add_argument("-m", dest="message", required=True)
    sub.add_parser("log")
    sub.add_parser("status")
    sub.add_parser("diff")
    r = sub.add_parser("restore")
    r.add_argument("name")
    return p

def main(argv):
    try:
        args = make_parser().parse_args(argv)
    except SystemExit as e:
        return e.code, []
    if args.command == "init":
        if os.path.exists(STORE):
            return 1, ["Already a snap folder"]
        os.makedirs(".snap", exist_ok=True)
        save({"commits": {}, "head": None})
        return 0, ["Initialised snap"]
    if not os.path.exists(STORE):
        return 1, ["Not a snap folder (run snap init)"]
    store = load()
    if args.command == "commit":
        files = working()
        if files == head_files(store):
            return 1, ["Nothing to commit"]
        cid = commit_id(files, args.message, store["head"])
        store["commits"][cid] = {"message": args.message, "parent": store["head"], "files": files}
        store["head"] = cid
        save(store)
        return 0, [f"[{cid}] {args.message}"]
    if args.command == "log":
        out, cid = [], store["head"]
        while cid:
            out.append(f"{cid} {store['commits'][cid]['message']}")
            cid = store["commits"][cid]["parent"]
        return 0, out or ["No commits yet"]
    if args.command == "status":
        out = [f"{kind}: {name}" for name, kind in changes(store)]
        return 0, out or ["Nothing changed"]
    if args.command == "diff":
        old, new = head_files(store), working()
        out = []
        for name, kind in changes(store):
            if kind == "modified":
                out.append(f"--- {name}")
                out += [l for l in difflib.ndiff(old[name].splitlines(), new[name].splitlines()) if l[:2] in ("- ", "+ ")]
        return 0, out
    old = head_files(store)
    if args.name not in old:
        return 1, [f"No such file in the last commit: {args.name}"]
    with open(args.name, "w") as f:
        f.write(old[args.name])
    return 0, [f"Restored {args.name}"]
:::

## Forge bonus: a pipeline

A shell pipe passes one command's output into the next: `cat notes.txt | grep egg | wc -l`. Each command is a small filter on lines of text. Build that idea in Python.

:::exercise id=forge2 points=30 title="Forge bonus: a pipeline"
Write `pipeline(line, files)` that runs a pipe of commands separated by `|` and returns the final output as a list of lines. `files` is a dictionary from file name to text. Each command takes the previous command's output lines as its input (the first gets `[]`):

- `cat NAME...`: the lines of those files (ignoring the input);
- `grep [-i] [-v] WORD`: input lines containing WORD (`-i` ignores case, `-v` keeps lines that **don't** contain it);
- `sort [-r] [-n]`: sorted (`-r` reversed, `-n` by number value);
- `uniq [-c]`: drops a line equal to the one before it; with `-c`, each line becomes `COUNT LINE` (count padded to width 4 with spaces on the left, then one space);
- `head [-n N]` and `tail [-n N]`: the first or last N lines (default 10);
- `wc -l`: one line holding the number of input lines.

Split each command's words with `shlex.split`. An unknown command raises `ValueError` with the message `unknown command: NAME`.
@@starter
import shlex

@@tests
FILES = {
    "fruit.txt": "apple\nBanana\napple\ncherry\nbanana\napple\n",
    "nums.txt": "10\n9\n100\n9\n",
}
def test_simple():
    "simple pipes"
    assert pipeline("cat fruit.txt | grep apple | wc -l", FILES) == ["3"]
    assert pipeline("cat fruit.txt | grep -i -v apple", FILES) == ["Banana", "cherry", "banana"]
    assert pipeline("cat nums.txt | sort -n -r | head -n 2", FILES) == ["100", "10"]
    assert pipeline("cat nums.txt | sort", FILES) == ["10", "100", "9", "9"]
def test_counting():
    "the word-count classic"
    assert pipeline("cat fruit.txt | sort | uniq -c | sort -r -n | head -n 1", FILES) == ["   3 apple"]
    assert pipeline("cat fruit.txt nums.txt | tail -n 2", FILES) == ["100", "9"]
    try:
        pipeline("cat fruit.txt | shout", FILES)
        assert False
    except ValueError as e:
        assert str(e) == "unknown command: shout"
@@hint
`sort -n` with `uniq -c` lines: sort by the number at the start, `float(line.split()[0])`. A dictionary from command names to small functions `(args, lines) -> lines` keeps it tidy.
@@solution
import shlex

def _opt(args, flag):
    return flag in args

def _n(args, default=10):
    return int(args[args.index("-n") + 1]) if "-n" in args else default

def pipeline(line, files):
    lines = []
    for part in line.split("|"):
        words = shlex.split(part)
        cmd, args = words[0], words[1:]
        if cmd == "cat":
            lines = [l for name in args for l in files[name].splitlines()]
        elif cmd == "grep":
            word = [a for a in args if not a.startswith("-")][0]
            ignore, invert = _opt(args, "-i"), _opt(args, "-v")
            def has(l):
                return (word.lower() in l.lower()) if ignore else (word in l)
            lines = [l for l in lines if has(l) != invert]
        elif cmd == "sort":
            key = (lambda l: float(l.split()[0])) if _opt(args, "-n") else None
            lines = sorted(lines, key=key, reverse=_opt(args, "-r"))
        elif cmd == "uniq":
            out, counts = [], []
            for l in lines:
                if out and out[-1] == l:
                    counts[-1] += 1
                else:
                    out.append(l)
                    counts.append(1)
            lines = [f"{c:>4} {l}" for c, l in zip(counts, out)] if _opt(args, "-c") else out
        elif cmd == "head":
            lines = lines[:_n(args)]
        elif cmd == "tail":
            n = _n(args)
            lines = lines[-n:] if n else []
        elif cmd == "wc":
            lines = [str(len(lines))]
        else:
            raise ValueError(f"unknown command: {cmd}")
    return lines
